  paintInlineImageXObject: function CanvasGraphics_paintInlineImageXObject(imgData) {
   var width = imgData.width;
   var height = imgData.height;

   // -------------------------------------------------------------
   // 【変更】ブラウザの物理限界（268,435,456 px）ぎりぎりまで引き上げ
   // -------------------------------------------------------------
   var maxPixels = 260000000; // 限界ぎりぎり（約2.6億画素）
   var maxDimension = 32767;  // 一辺のハードウェア上限
   var currentPixels = width * height;

   if (currentPixels > maxPixels || width > maxDimension || height > maxDimension) {
     var downscaleFactor = Math.min(
       Math.sqrt(maxPixels / currentPixels),
       maxDimension / Math.max(width, height)
     );
     var targetWidth = Math.floor(width * downscaleFactor);
     var targetHeight = Math.floor(height * downscaleFactor);

     var ctx = this.ctx;
     this.save();
     ctx.scale(1 / width, -1 / height);

     // 限界ぎりぎりの縮小サイズで Canvas を確保
     var tmpCanvas = this.cachedCanvases.getCanvas('inlineImageClamped', targetWidth, targetHeight);
     var tmpCtx = tmpCanvas.context;

     var clampedImageData = tmpCtx.createImageData(targetWidth, targetHeight);
     var srcData = imgData.data;
     var dstData = clampedImageData.data;

     var scaleX = width / targetWidth;
     var scaleY = height / targetHeight;

     if (imgData.kind === ImageKind.RGBA_32BPP || !imgData.kind) {
       for (var y = 0; y < targetHeight; y++) {
         var srcY = Math.floor(y * scaleY);
         var srcRowOffset = srcY * width * 4;
         var dstRowOffset = y * targetWidth * 4;
         for (var x = 0; x < targetWidth; x++) {
           var srcX = Math.floor(x * scaleX);
           var srcIdx = srcRowOffset + srcX * 4;
           var dstIdx = dstRowOffset + x * 4;
           dstData[dstIdx] = srcData[srcIdx];
           dstData[dstIdx + 1] = srcData[srcIdx + 1];
           dstData[dstIdx + 2] = srcData[srcIdx + 2];
           dstData[dstIdx + 3] = srcData[srcIdx + 3];
         }
       }
     } else if (imgData.kind === ImageKind.RGB_24BPP) {
       for (var y = 0; y < targetHeight; y++) {
         var srcY = Math.floor(y * scaleY);
         var srcRowOffset = srcY * width * 3;
         var dstRowOffset = y * targetWidth * 4;
         for (var x = 0; x < targetWidth; x++) {
           var srcX = Math.floor(x * scaleX);
           var srcIdx = srcRowOffset + srcX * 3;
           var dstIdx = dstRowOffset + x * 4;
           dstData[dstIdx] = srcData[srcIdx];
           dstData[dstIdx + 1] = srcData[srcIdx + 1];
           dstData[dstIdx + 2] = srcData[srcIdx + 2];
           dstData[dstIdx + 3] = 255;
         }
       }
     } else if (imgData.kind === ImageKind.GRAYSCALE_1BPP) {
       var fullSrcDiff = width + 7 >> 3;
       for (var y = 0; y < targetHeight; y++) {
         var srcY = Math.floor(y * scaleY);
         var srcRowOffset = srcY * fullSrcDiff;
         var dstRowOffset = y * targetWidth * 4;
         for (var x = 0; x < targetWidth; x++) {
           var srcX = Math.floor(x * scaleX);
           var byteIdx = srcRowOffset + (srcX >> 3);
           var bitMask = 128 >> (srcX & 7);
           var val = (srcData[byteIdx] & bitMask) ? 255 : 0;
           var dstIdx = dstRowOffset + x * 4;
           dstData[dstIdx] = val;
           dstData[dstIdx + 1] = val;
           dstData[dstIdx + 2] = val;
           dstData[dstIdx + 3] = 255;
         }
       }
     }

     tmpCtx.putImageData(clampedImageData, 0, 0);

     ctx.drawImage(tmpCanvas.canvas, 0, 0, targetWidth, targetHeight, 0, -height, width, height);
     this.restore();
     return;
   }
   // -------------------------------------------------------------
   var ctx = this.ctx;
   this.save();
   ctx.scale(1 / width, -1 / height);
   var currentTransform = ctx.mozCurrentTransformInverse;
   var a = currentTransform[0], b = currentTransform[1];
   var widthScale = Math.max(Math.sqrt(a * a + b * b), 1);
   var c = currentTransform[2], d = currentTransform[3];
   var heightScale = Math.max(Math.sqrt(c * c + d * d), 1);
   var imgToPaint, tmpCanvas;
   if (imgData instanceof HTMLElement || !imgData.data) {
    imgToPaint = imgData;
   } else {
    tmpCanvas = this.cachedCanvases.getCanvas('inlineImage', width, height);
    var tmpCtx = tmpCanvas.context;
    putBinaryImageData(tmpCtx, imgData);
    imgToPaint = tmpCanvas.canvas;
   }
   var paintWidth = width, paintHeight = height;
   var tmpCanvasId = 'prescale1';
   while (widthScale > 2 && paintWidth > 1 || heightScale > 2 && paintHeight > 1) {
    var newWidth = paintWidth, newHeight = paintHeight;
    if (widthScale > 2 && paintWidth > 1) {
     newWidth = Math.ceil(paintWidth / 2);
     widthScale /= paintWidth / newWidth;
    }
    if (heightScale > 2 && paintHeight > 1) {
     newHeight = Math.ceil(paintHeight / 2);
     heightScale /= paintHeight / newHeight;
    }
    tmpCanvas = this.cachedCanvases.getCanvas(tmpCanvasId, newWidth, newHeight);
    tmpCtx = tmpCanvas.context;
    tmpCtx.clearRect(0, 0, newWidth, newHeight);
    tmpCtx.drawImage(imgToPaint, 0, 0, paintWidth, paintHeight, 0, 0, newWidth, newHeight);
    imgToPaint = tmpCanvas.canvas;
    paintWidth = newWidth;
    paintHeight = newHeight;
    tmpCanvasId = tmpCanvasId === 'prescale1' ? 'prescale2' : 'prescale1';
   }
   ctx.drawImage(imgToPaint, 0, 0, paintWidth, paintHeight, 0, -height, width, height);
   if (this.imageLayer) {
    var position = this.getCanvasPosition(0, -height);
    this.imageLayer.appendImage({
     imgData: imgData,
     left: position[0],
     top: position[1],
     width: width / currentTransform[0],
     height: height / currentTransform[3]
    });
   }
   this.restore();
  },
